rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function userExists() { return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function userDoc() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    
    function isTeacher() {
      return signedIn() && (
        (request.auth.token.email != null && request.auth.token.email.lower() == 'trntui6@gmail.com') ||
        (userExists() && 'role' in userDoc() && userDoc().role == 'teacher')
      );
    }
    function isTeacherEmail() {
      return signedIn() && request.auth.token.email != null && request.auth.token.email.lower() == 'trntui6@gmail.com';
    }
    function isVerified() {
      return signedIn() && (isTeacher() || (userExists() && 'emailVerified' in userDoc() && userDoc().emailVerified == true) || (request.auth.token.email_verified != null && request.auth.token.email_verified == true));
    }
    function isHiddenUser() {
      return signedIn() && (request.auth.token.email == null || request.auth.token.email.lower() != 'trntui6@gmail.com') && userExists() && ('hidden' in userDoc()) && userDoc().hidden == true;
    }
    function isActive() { return signedIn() && userExists() && !isHiddenUser(); }
    function only(keys) { return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys); }
    function validClass(c) { return c == '' || c.matches('^(10|11|12)(a([1-9]|[1-9][0-9]))?$'); }
    function isNullOrMissing(data, key) { return !data.keys().hasAny([key]) || data[key] == null; }

    // QUẢN LÝ USER (CẬP NHẬT QUAN TRỌNG)
    match /users/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || isTeacher());
      allow list: if isTeacher();
      
      // Tạo mới: Có thể thêm infoLocked: false mặc định
      allow create: if signedIn() && request.auth.uid == uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.role == (isTeacherEmail() ? 'teacher' : 'student')
        && request.resource.data.emailVerified == request.auth.token.email_verified
        && validClass(request.resource.data.class);
        
      // Cập nhật: 
      allow update: if isTeacher()
        || (isActive() && request.auth.uid == uid
            && only(['class', 'name', 'updatedAt']) // HS chỉ được đổi tên và lớp
            && validClass(request.resource.data.class)
            // 🟢 CHỐT CHẶN: Nếu infoLocked == true thì chặn không cho HS sửa name/class
            && (!('infoLocked' in resource.data) || resource.data.infoLocked == false)
           );
           
      allow delete: if isTeacher();
    }

    // ... (CÁC COLLECTION KHÁC GIỮ NGUYÊN NHƯ CŨ)
    match /exams/{id} { allow read: if isActive() || isTeacher(); allow write: if isTeacher(); }
    match /exam_keys/{id} { allow read, write: if isTeacher(); }
    match /exam_assignments/{id} { allow read: if isActive() || isTeacher(); allow write, delete: if isTeacher(); }
    match /exam_submissions/{id} {
      allow get, list: if isTeacher() || (isActive() && resource.data.studentId == request.auth.uid);
      allow create: if isActive() && isVerified() && request.resource.data.studentId == request.auth.uid && request.resource.data.status in ['in_progress', 'submitted'] && request.resource.data.score == null && request.resource.data.isPublished == false;
      allow update: if isTeacher() || (isActive() && resource.data.studentId == request.auth.uid && resource.data.status == 'in_progress' && request.resource.data.status == 'submitted');
      allow delete: if isTeacher();
    }
    match /exam_rosters/{classId} { allow read: if isActive() || isTeacher(); allow write: if isTeacher(); }
    match /vl10_questions/{document} { allow read: if signedIn(); allow write: if isTeacher(); }
    match /assignments/{id} { allow read: if isActive() || isTeacher(); allow write: if isTeacher(); match /answerKey/{keyId} { allow read, write: if isTeacher(); } }
    match /submissions/{id} { allow get, list: if isTeacher() || (isActive() && resource.data.studentId == request.auth.uid); allow create: if isActive() && isVerified() && request.resource.data.studentId == request.auth.uid && request.resource.data.status == 'in_progress' && isNullOrMissing(request.resource.data, 'score'); allow update: if isTeacher() || (isActive() && resource.data.studentId == request.auth.uid); allow delete: if isTeacher(); }
    match /viewLogs/{id} { allow get, list: if isTeacher() || (isActive() && resource.data.studentId == request.auth.uid); allow create: if isActive() && isVerified() && request.resource.data.studentId == request.auth.uid; allow update, delete: if isTeacher() || (isActive() && resource.data.studentId == request.auth.uid); }
    match /notifications/{id} { allow read: if isActive() || isTeacher(); allow create, update, delete: if isTeacher(); }
    match /rosters/{classId} { allow read: if isActive() || isTeacher(); allow write: if isTeacher(); }
    match /auditLogs/{id} { allow read, create: if isTeacher(); allow update, delete: if false; }
    match /settings/{docId} { allow read: if signedIn(); allow write: if isTeacher(); }
    match /lessons/{id} { allow read: if signedIn(); allow write: if isTeacher(); }

    match /{document=**} { allow read, write: if false; }
  }
}
