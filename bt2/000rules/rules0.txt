rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ═══════════════════════════════════════════════════
    // USERS
    // ═══════════════════════════════════════════════════
    match /users/{userId} {
      allow read: if request.auth != null;
      allow create: if request.auth.uid == userId;
      allow update, delete: if request.auth.uid == userId
        || request.auth.token.email == "trntui6@gmail.com";
    }
    
    // ═══════════════════════════════════════════════════
    // SETTINGS
    // ═══════════════════════════════════════════════════
    match /settings/{doc} {
      allow read: if request.auth != null;
      allow write: if request.auth.token.email == "trntui6@gmail.com";
    }
    
    // ═══════════════════════════════════════════════════
    // LESSONS
    // ═══════════════════════════════════════════════════
    match /lessons/{doc} {
      allow read: if request.auth != null;
      allow write: if request.auth.token.email == "trntui6@gmail.com";
    }
    
    // ═══════════════════════════════════════════════════
    // ASSIGNMENTS
    // ═══════════════════════════════════════════════════
    match /assignments/{doc} {
      allow read: if request.auth != null;
      allow write: if request.auth.token.email == "trntui6@gmail.com";
    }
    
    // ═══════════════════════════════════════════════════
    // SUBMISSIONS (bài làm HS) — ĐÃ SIẾT
    // ═══════════════════════════════════════════════════
    match /submissions/{subId} {
      // HS chỉ đọc bài CỦA MÌNH, GV đọc tất cả
      allow read: if request.auth != null && (
        resource.data.studentId == request.auth.uid ||
        request.auth.token.email == "trntui6@gmail.com"
      );
      
      // HS chỉ tạo bài cho CHÍNH MÌNH
      allow create: if request.auth != null && 
        request.resource.data.studentId == request.auth.uid;
      
      // HS chỉ sửa bài CỦA MÌNH, GV sửa tất cả
      allow update: if request.auth != null && (
        resource.data.studentId == request.auth.uid ||
        request.auth.token.email == "trntui6@gmail.com"
      );
      
      // Chỉ GV xoá
      allow delete: if request.auth.token.email == "trntui6@gmail.com";
    }
    
    // ═══════════════════════════════════════════════════
    // VIEWLOGS (log truy cập) — ĐÃ SIẾT
    // ═══════════════════════════════════════════════════
    match /viewLogs/{logId} {
      // HS chỉ đọc log CỦA MÌNH, GV đọc tất cả
      allow read: if request.auth != null && (
        resource.data.studentId == request.auth.uid ||
        request.auth.token.email == "trntui6@gmail.com"
      );
      
      // HS tạo log mới (khi mở file)
      allow create: if request.auth != null;
      
      // HS chỉ sửa log CỦA MÌNH, GV sửa tất cả
      allow update: if request.auth != null && (
        resource.data.studentId == request.auth.uid ||
        request.auth.token.email == "trntui6@gmail.com"
      );
      
      // Chỉ GV xoá
      allow delete: if request.auth.token.email == "trntui6@gmail.com";
    }
    
    // ═══════════════════════════════════════════════════
    // NOTIFICATIONS
    // ═══════════════════════════════════════════════════
    match /notifications/{doc} {
      allow read: if request.auth != null;
      allow write: if request.auth.token.email == "trntui6@gmail.com";
    }
    
    // ═══════════════════════════════════════════════════
    // DELETELOGS (Lịch sử xoá tài khoản)
    // ═══════════════════════════════════════════════════
    match /deleteLogs/{logId} {
      allow read: if request.auth.token.email == "trntui6@gmail.com";
      allow create: if request.auth.token.email == "trntui6@gmail.com";
      allow update, delete: if false;
    }
    
  }
}
